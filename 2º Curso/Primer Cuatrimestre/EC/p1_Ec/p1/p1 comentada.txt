; ----------------------------------------------------------
;  PROGRAMA 8086 — BINARIO NATURAL Y COMPLEMENTO A 1
;  Siempre muestra:
;     → Valor en binario natural
;     → Valor en complemento a 1
;
;  Solo se invierten los bits si el primer bit es 1.
; ----------------------------------------------------------

.MODEL SMALL           ; Modelo SMALL: 1 segmento código + 1 de datos
.STACK 100h            ; Reserva 256 bytes para la pila

.DATA                  ; Inicio del segmento de datos

; Estructura usada por INT 21h / función 0Ah
; CADENA[0] = Nº máx de caracteres a leer (4 para binario + CR = 5)
; CADENA[1] = Nº real de caracteres leídos (lo rellena DOS)
; CADENA[2..] = caracteres introducidos por el usuario
CADENA DB 5,0,0,0,0,0,0

; Tabla de pesos para convertir binario → decimal
; bit0 = 8, bit1 = 4, bit2 = 2, bit3 = 1
PESO   DB 8,4,2,1

; Variables donde guardar los resultados en decimal
DEC_NAT  DB ?
DEC_COMP DB ?

; Mensajes
MSG1    DB 13,10,'Introduce un numero binario de 4 bits: $'
MSGNAT  DB 13,10,'Valor en binario natural: $'
MSGCOMP DB 13,10,'Valor en complemento a 1: $'

; ----------------------------------------------------------
.CODE               ; Inicio de segmento de código
; ----------------------------------------------------------

    ; Inicializar el registro de segmento DS con el segmento de datos
    MOV AX, @DATA
    MOV DS, AX

    ; Mostrar mensaje de entrada
    MOV AH, 09h
    LEA DX, MSG1
    INT 21h

    ; Leer cadena por teclado (INT 21h, función 0Ah)
    MOV AH, 0Ah
    LEA DX, CADENA
    INT 21h

    ; Convertir caracteres ASCII ('0' y '1') → números 0 y 1
    SUB CADENA[2], 48
    SUB CADENA[3], 48
    SUB CADENA[4], 48
    SUB CADENA[5], 48

; ==========================================================
; 1) CALCULAR SIEMPRE EL VALOR NATURAL
; ==========================================================

    ; bit0 * 8
    MOV AL, CADENA[2]
    MUL PESO[0]
    MOV DEC_NAT, AL

    ; bit1 * 4
    MOV AL, CADENA[3]
    MUL PESO[1]
    ADD DEC_NAT, AL

    ; bit2 * 2
    MOV AL, CADENA[4]
    MUL PESO[2]
    ADD DEC_NAT, AL

    ; bit3 * 1
    MOV AL, CADENA[5]
    MUL PESO[3]
    ADD DEC_NAT, AL

; ==========================================================
; 2) DECIDIR SI INVERTIR O NO (complemento a 1)
;    Si primer bit = 1 → invertir
;    Si primer bit = 0 → NO invertir
; ==========================================================

    MOV AL, CADENA[2] ; cargar primer bit
    CMP AL, 1         ; ¿es 1?
    JNE NO_INVERTIR   ; si NO es 1 → saltar

; ----------------------------------------------------------
; INVERSIÓN DE BITS (COMPLEMENTO A 1) SOLO SI MSB = 1
; ----------------------------------------------------------
    MOV BL, CADENA[2]
    NOT BL
    AND BL, 1
    MOV CADENA[2], BL

    MOV BL, CADENA[3]
    NOT BL
    AND BL, 1
    MOV CADENA[3], BL

    MOV BL, CADENA[4]
    NOT BL
    AND BL, 1
    MOV CADENA[4], BL

    MOV BL, CADENA[5]
    NOT BL
    AND BL, 1
    MOV CADENA[5], BL

    JMP CALC_COMP   ; saltar directamente a calcular decimal

; ----------------------------------------------------------
; SI NO HAY QUE INVERTIR (primer bit = 0)
; simplemente dejamos los bits tal cual
; ----------------------------------------------------------
NO_INVERTIR:
    MOV AL, CADENA[2]
    MOV CADENA[2], AL

    MOV AL, CADENA[3]
    MOV CADENA[3], AL

    MOV AL, CADENA[4]
    MOV CADENA[4], AL

    MOV AL, CADENA[5]
    MOV CADENA[5], AL

; ==========================================================
; 3) CALCULAR DECIMAL DEL COMPLEMENTO A 1
; ==========================================================
CALC_COMP:
    ; bit0 * 8
    MOV AL, CADENA[2]
    MUL PESO[0]
    MOV DEC_COMP, AL

    ; bit1 * 4
    MOV AL, CADENA[3]
    MUL PESO[1]
    ADD DEC_COMP, AL

    ; bit2 * 2
    MOV AL, CADENA[4]
    MUL PESO[2]
    ADD DEC_COMP, AL

    ; bit3 * 1
    MOV AL, CADENA[5]
    MUL PESO[3]
    ADD DEC_COMP, AL

; ==========================================================
; MOSTRAR VALOR EN NATURAL
; ==========================================================
    MOV AH, 09h
    LEA DX, MSGNAT
    INT 21h

    MOV AL, DEC_NAT
    CALL PRINT_DEC

; ==========================================================
; MOSTRAR VALOR EN COMPLEMENTO A 1
; ==========================================================
    MOV AH, 09h
    LEA DX, MSGCOMP
    INT 21h

    MOV AL, DEC_COMP
    CALL PRINT_DEC

; ==========================================================
; FIN DEL PROGRAMA
; ==========================================================
    MOV AH, 00h     ; Esperar tecla
    INT 16h
    MOV AH, 4Ch     ; Salir al DOS
    INT 21h

; ----------------------------------------------------------
; SUBRUTINA PARA IMPRIMIR NÚMEROS DECIMALES (0–15)
; ----------------------------------------------------------
PRINT_DEC PROC
    MOV AH,0
    MOV BL,10      ; divisor = 10
    DIV BL         ; AL = decenas, AH = unidades

    CMP AL,0
    JE DIG_UNO     ; si AL=0 → no imprimir decenas

    ADD AL,48      ; convertir decenas a ASCII
    MOV DL,AL
    MOV AH,02h
    INT 21h

DIG_UNO:
    ADD AH,48      ; convertir unidades a ASCII
    MOV DL,AH
    MOV AH,02h
    INT 21h
    RET
PRINT_DEC ENDP

END
